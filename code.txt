    "use strict";

    /*
      CONFIGURAÇÃO

      WHATSAPP:
      Informe um número real e autorizado, somente com dígitos,
      incluindo país e DDD. Deixe vazio para manter a demonstração.

      ESF_URL:
      Informe a URL HTTPS do seu sistema interno real.

    const CONFIG = Object.freeze({
      WHATSAPP: "",
      ESF_URL: ""
    });

    document.documentElement.classList.add("js");

    // Menu móvel
    const menuButton = document.querySelector(".menu-button");
    const menu = document.querySelector("#menu");

    function closeMenu(restoreFocus = false) {
      const wasOpen = menuButton.getAttribute("aria-expanded") === "true";
      menu.classList.remove("open");
      menuButton.setAttribute("aria-expanded", "false");

      if (wasOpen && restoreFocus) {
        menuButton.focus();
      }
    }

    menuButton.addEventListener("click", () => {
      const open = menuButton.getAttribute("aria-expanded") !== "true";
      menuButton.setAttribute("aria-expanded", String(open));
      menu.classList.toggle("open", open);
    });

    menu.querySelectorAll("a").forEach((link) => {
      link.addEventListener("click", () => closeMenu());
    });

    document.addEventListener("click", (event) => {
      if (!event.target.closest(".header")) {
        closeMenu();
      }
    });

    document.addEventListener("keydown", (event) => {
      if (event.key === "Escape") {
        closeMenu(true);
      }
    });

    window.matchMedia("(max-width: 800px)").addEventListener("change", () => {
      closeMenu();
    });

    // Fotos opcionais. O conteúdo ilustrado permanece se não existirem.
    [
      ["assets/hero.png", "#hero-art"],
      ["assets/team.png", "#team-art"]
    ].forEach(([source, selector]) => {
      const image = new Image();

      image.onload = () => {
        document.querySelector(selector).classList.add("has-photo");
      };

      image.src = source;
    });

    // Janela de revisão/demonstração
    const dialog = document.querySelector("#info-dialog");
    const dialogTitle = document.querySelector("#dialog-title");
    const dialogDescription = document.querySelector("#dialog-description");
    const dialogSummary = document.querySelector("#dialog-summary");
    const dialogNote = document.querySelector("#dialog-note");
    const externalLink = document.querySelector("#external-link");

    let previousFocus = null;
    let previousOverflow = "";

    function openDialog({
      title,
      description,
      summary = "",
      note,
      url = "",
      linkLabel = ""
    }) {
      previousFocus = document.activeElement;

      dialogTitle.textContent = title;
      dialogDescription.textContent = description;
      dialogSummary.textContent = summary;
      dialogSummary.hidden = !summary;
      dialogNote.textContent = note;

      externalLink.hidden = !url;
      externalLink.textContent = linkLabel;

      if (url) {
        externalLink.href = url;
      } else {
        externalLink.removeAttribute("href");
      }

      previousOverflow = document.body.style.overflow;
      document.body.style.overflow = "hidden";

      dialog.showModal();
      document.querySelector(".dialog-close").focus();
    }

    document.querySelectorAll(".dialog-close, .dialog-dismiss").forEach((button) => {
      button.addEventListener("click", () => dialog.close());
    });

    dialog.addEventListener("close", () => {
      document.body.style.overflow = previousOverflow;

      if (previousFocus && previousFocus.isConnected) {
        previousFocus.focus();
      }
    });

    dialog.addEventListener("click", (event) => {
      if (event.target !== dialog) return;

      const bounds = dialog.getBoundingClientRect();

      if (
        event.clientX < bounds.left ||
        event.clientX > bounds.right ||
        event.clientY < bounds.top ||
        event.clientY > bounds.bottom
      ) {
        dialog.close();
      }
    });

    function hasWhatsApp() {
      return /^[1-9]\d{7,14}$/.test(CONFIG.WHATSAPP);
    }

    function showContact(message) {
      const configured = hasWhatsApp();

      openDialog({
        title: configured
          ? "Revise sua mensagem"

          ? "A corrida depende da confirmação da central. Abrir o WhatsApp não confirma uma reserva."

        url: configured
          ? "https://wa.me/" + CONFIG.WHATSAPP +
            "?text=" + encodeURIComponent(message)
          : "",
        linkLabel: "Continuar no WhatsApp ↗"
      });
    }

      button.addEventListener("click", () => {
        showContact(button.dataset.contact);

    // Formulário de trajeto
    const form = document.querySelector("#ride-form");
    const origin = document.querySelector("#origin");
    const destination = document.querySelector("#destination");

    const scheduleFields = document.querySelector("#schedule-fields");
    const rideDate = document.querySelector("#ride-date");
    const rideTime = document.querySelector("#ride-time");

    function saoPauloNow() {
      const parts = new Intl.DateTimeFormat("en-CA", {
        timeZone: "America/Sao_Paulo",
        year: "numeric",
        month: "2-digit",
        day: "2-digit",
        hour: "2-digit",
        minute: "2-digit",
        hourCycle: "h23"
      }).formatToParts(new Date());

      const value = (type) => parts.find((part) => part.type === type).value;

      return {
        date: value("year") + "-" + value("month") + "-" + value("day"),
        time: value("hour") + ":" + value("minute")
      };
    }

    function updateSchedule() {
      const scheduled = when.value === "scheduled";
      const now = saoPauloNow();

      scheduleFields.hidden = !scheduled;
      rideDate.required = scheduled;
      rideTime.required = scheduled;

      // Campos inativos não participam da validação.
      rideDate.disabled = !scheduled;
      rideTime.disabled = !scheduled;

      rideDate.min = now.date;
      rideTime.min = rideDate.value === now.date ? now.time : "";

      rideDate.setCustomValidity("");
      rideTime.setCustomValidity("");
    }

    when.addEventListener("change", updateSchedule);
    rideDate.addEventListener("change", updateSchedule);
    rideTime.addEventListener("input", () => rideTime.setCustomValidity(""));

    [origin, destination].forEach((field) => {
      field.addEventListener("input", () => field.setCustomValidity(""));
    });

    document.querySelectorAll("[data-schedule]").forEach((link) => {
      link.addEventListener("click", () => {
        when.value = "scheduled";
        updateSchedule();
      });
    });

    form.addEventListener("submit", (event) => {
      event.preventDefault();

      [origin, destination].forEach((field) => {
        field.setCustomValidity(
          field.value.trim() ? "" : "Informe um local válido."
        );
      });

      let requestedTime = "Agora";

      if (when.value === "scheduled") {
        const now = saoPauloNow();
        const selected = rideDate.value + "T" + rideTime.value;
        const current = now.date + "T" + now.time;

        rideTime.setCustomValidity(
          selected <= current
            ? "Escolha uma data e um horário futuros em São Paulo."
            : ""
        );

        if (rideDate.value && rideTime.value) {
          const [year, month, day] = rideDate.value.split("-");
          requestedTime =
            day + "/" + month + "/" + year +
            " às " + rideTime.value + " (São Paulo)";
        }
      }

      if (!form.reportValidity()) return;

      const message = [
        "Olá! Gostaria de solicitar uma corrida.",
        "",
        "Partida: " + origin.value.trim(),
        "Destino: " + destination.value.trim(),
        "Quando: " + requestedTime,
        "",
        "Aguardo a confirmação de disponibilidade pela central."
      ].join("\n");

      showContact(message);
    });

    updateSchedule();

        const parsed = new URL(CONFIG.ESF_URL);

          : "Nenhum sistema interno está conectado. Não informe senhas nesta demonstração.",
        url,
        linkLabel: "Abrir sistema ESF ↗"
      });
    });
